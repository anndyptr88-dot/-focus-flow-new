// Firebase Storage Rules untuk foto profil Focus Flow.
// Masukkan aturan ini melalui Firebase Console > Storage > Rules.
// Pengguna hanya dapat mengunggah/mengubah file di folder UID miliknya.

 rules_version = '2';
service firebase.storage {
  match /b/{bucket}/o {
    match /profilePhotos/{userId}/{fileName} {
      allow read: if request.auth != null;
      allow write: if request.auth != null
                    && request.auth.uid == userId
                    && request.resource.size < 5 * 1024 * 1024
                    && request.resource.contentType.matches('image/.*');
    }
  }
}
